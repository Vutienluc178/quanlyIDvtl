Hướng dẫn cài đặt nhanh API Google Sheets:

  1. Mở file Google Sheets, trên thanh menu chọn Tiện ích mở rộng -> Apps Script.
  2. Xóa mã cũ, sao chép và dán toàn bộ đoạn mã bên dưới vào.
  3. Bấm Triển khai (Deploy) -> Tùy chọn triển khai mới (New deployment).
  4. Chọn loại Ứng dụng web (Web App), Quyền truy cập chọn Bất kỳ ai (Anyone) rồi bấm Triển khai.
  5. Copy URL Ứng dụng web và dán vào ô bên trên.
    Lưu ý quan trọng: Nếu bạn sửa đổi code này, khi lưu phải tạo "Phiên bản mới (New version)" thì URL mới nhận code mới.
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheets()[0]; 
    
    var data = JSON.parse(e.postData.contents); 
    
    var rows = [];
    for (var i = 0; i < data.length; i++) {
      rows.push([
        data[i].id || "",       // Cột A: ID
        data[i].content || ""   // Cột B: Toàn bộ nội dung
      ]);
    }
    
    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
